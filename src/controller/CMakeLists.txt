cmake_minimum_required(VERSION 3.18)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(crossstage_controller LANGUAGES CXX)
  enable_testing()
endif()

get_filename_component(STITCH_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT TARGET stitch_common)
  add_library(stitch_common INTERFACE)
  target_include_directories(stitch_common INTERFACE "${STITCH_REPO_ROOT}/src")
endif()

add_library(stitch_controller STATIC controller.cpp v5.cpp policy.cpp validation.cpp reuse.cpp)
target_link_libraries(stitch_controller PUBLIC stitch_common)
target_compile_features(stitch_controller PUBLIC cxx_std_17)

add_executable(test_controller "${STITCH_REPO_ROOT}/tests/controller/test_controller.cpp")
target_include_directories(test_controller PRIVATE "${STITCH_REPO_ROOT}/tests")
target_link_libraries(test_controller PRIVATE stitch_controller)
add_test(NAME test_controller COMMAND test_controller)

add_executable(controller_trace "${STITCH_REPO_ROOT}/tests/controller/controller_trace.cpp")
target_link_libraries(controller_trace PRIVATE stitch_controller)
add_test(NAME controller_trace COMMAND controller_trace)
add_executable(test_learned_controller "${STITCH_REPO_ROOT}/tests/controller/test_learned_controller.cpp")
target_include_directories(test_learned_controller PRIVATE "${STITCH_REPO_ROOT}/tests")
target_link_libraries(test_learned_controller PRIVATE stitch_controller)
add_test(NAME test_learned_controller COMMAND test_learned_controller)
add_executable(test_ensemble "${STITCH_REPO_ROOT}/tests/controller/test_ensemble.cpp")
target_include_directories(test_ensemble PRIVATE "${STITCH_REPO_ROOT}/tests")
target_link_libraries(test_ensemble PRIVATE stitch_controller)
add_test(NAME test_ensemble COMMAND test_ensemble)

option(STITCH_CONTROLLER_GPU_TESTS "Test Controller with actual S5 CUDA feedback" OFF)
if(STITCH_CONTROLLER_GPU_TESTS)
  include(CheckLanguage)
  check_language(CUDA)
  if(NOT CMAKE_CUDA_COMPILER)
    message(FATAL_ERROR "Controller/S5 integration tests require a CUDA toolkit")
  endif()
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 75)
  endif()
  enable_language(CUDA)
  if(NOT TARGET stitch_s5)
    add_subdirectory("${STITCH_REPO_ROOT}/src/s5" s5)
  endif()
  add_executable(test_controller_s5 "${STITCH_REPO_ROOT}/tests/controller/test_controller_s5.cu")
  target_include_directories(test_controller_s5 PRIVATE "${STITCH_REPO_ROOT}/tests")
  target_link_libraries(test_controller_s5 PRIVATE stitch_controller stitch_s5)
  set_target_properties(test_controller_s5 PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_test(NAME test_controller_s5 COMMAND test_controller_s5)
  set_tests_properties(test_controller_s5 PROPERTIES SKIP_RETURN_CODE 77)
endif()

add_executable(test_v5 "${STITCH_REPO_ROOT}/tests/controller/test_v5.cpp")
target_include_directories(test_v5 PRIVATE "${STITCH_REPO_ROOT}/tests")
target_link_libraries(test_v5 PRIVATE stitch_controller)
add_test(NAME test_v5 COMMAND test_v5)
