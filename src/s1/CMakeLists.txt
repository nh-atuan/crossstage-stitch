cmake_minimum_required(VERSION 3.18)

# Standalone stage build; shared root build files remain owned by Tuan.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(crossstage_s1 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  enable_testing()
endif()

get_filename_component(STITCH_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT TARGET stitch_common)
  add_library(stitch_common INTERFACE)
  target_include_directories(stitch_common INTERFACE "${STITCH_REPO_ROOT}/src")
endif()

include(CheckLanguage)
check_language(CUDA)
if(CMAKE_CUDA_COMPILER)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 75)
  endif()
  enable_language(CUDA)
  add_library(stitch_s1 STATIC s1.cu)
  target_link_libraries(stitch_s1 PUBLIC stitch_common)
  set_target_properties(stitch_s1 PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_executable(test_s1_cuda "${STITCH_REPO_ROOT}/tests/test_s1_cuda.cu")
  target_include_directories(test_s1_cuda PRIVATE "${STITCH_REPO_ROOT}/tests")
  target_link_libraries(test_s1_cuda PRIVATE stitch_s1)
  set_target_properties(test_s1_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_test(NAME test_s1_cuda COMMAND test_s1_cuda)
  set_tests_properties(test_s1_cuda PROPERTIES SKIP_RETURN_CODE 77)
else()
  message(STATUS "S1: no CUDA compiler; CUDA targets skipped")
endif()
