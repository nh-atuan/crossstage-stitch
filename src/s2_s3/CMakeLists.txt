cmake_minimum_required(VERSION 3.18)

# Standalone stage build; shared root build files remain owned by Tuan.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(crossstage_s2_s3 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  enable_testing()
endif()

get_filename_component(STITCH_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT TARGET stitch_common)
  add_library(stitch_common INTERFACE)
  target_include_directories(stitch_common INTERFACE "${STITCH_REPO_ROOT}/src")
endif()

add_library(stitch_s2_s3_config STATIC s2_s3_config.cpp)
target_link_libraries(stitch_s2_s3_config PUBLIC stitch_common)
target_compile_features(stitch_s2_s3_config PUBLIC cxx_std_17)
add_executable(test_s2_s3_config "${STITCH_REPO_ROOT}/tests/test_s2_s3_config.cpp")
target_include_directories(test_s2_s3_config PRIVATE "${STITCH_REPO_ROOT}/tests")
target_link_libraries(test_s2_s3_config PRIVATE stitch_s2_s3_config)
add_test(NAME test_s2_s3_config COMMAND test_s2_s3_config)

include(CheckLanguage)
check_language(CUDA)
if(CMAKE_CUDA_COMPILER)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 75)
  endif()
  enable_language(CUDA)
  add_library(stitch_s2_s3 STATIC s2.cu s3.cu)
  target_link_libraries(stitch_s2_s3 PUBLIC stitch_s2_s3_config)
  set_target_properties(stitch_s2_s3 PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_executable(test_s2_s3_cuda "${STITCH_REPO_ROOT}/tests/test_s2_s3_cuda.cu")
  target_include_directories(test_s2_s3_cuda PRIVATE "${STITCH_REPO_ROOT}/tests")
  target_link_libraries(test_s2_s3_cuda PRIVATE stitch_s2_s3)
  set_target_properties(test_s2_s3_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_test(NAME test_s2_s3_cuda COMMAND test_s2_s3_cuda)
  set_tests_properties(test_s2_s3_cuda PROPERTIES SKIP_RETURN_CODE 77)
  # Ablation A3 benchmark; the CTest entry only runs its synthetic self-test.
  add_executable(s2_s3_a3_bench "${STITCH_REPO_ROOT}/tests/s2_s3/a3_budget_bench.cu")
  target_include_directories(s2_s3_a3_bench PRIVATE "${STITCH_REPO_ROOT}/tests")
  target_link_libraries(s2_s3_a3_bench PRIVATE stitch_s2_s3)
  set_target_properties(s2_s3_a3_bench PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_test(NAME s2_s3_a3_bench_smoke COMMAND s2_s3_a3_bench --self-test)
  set_tests_properties(s2_s3_a3_bench_smoke PROPERTIES SKIP_RETURN_CODE 77)
else()
  message(STATUS "S2/S3: no CUDA compiler; only host configuration tests are built")
endif()
