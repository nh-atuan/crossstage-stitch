cmake_minimum_required(VERSION 3.18)

# Standalone, like S6: shared root build files remain owned by Tuan.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(crossstage_s4_s5 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  enable_testing()
endif()

get_filename_component(STITCH_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT TARGET stitch_common)
  add_library(stitch_common INTERFACE)
  target_include_directories(stitch_common INTERFACE "${STITCH_REPO_ROOT}/src")
endif()

add_library(stitch_s4_config STATIC s4_config.cpp)
target_link_libraries(stitch_s4_config PUBLIC stitch_common)
target_compile_features(stitch_s4_config PUBLIC cxx_std_17)
add_library(stitch_s5_config STATIC s5_config.cpp)
target_link_libraries(stitch_s5_config PUBLIC stitch_common)
target_compile_features(stitch_s5_config PUBLIC cxx_std_17)

foreach(stage_test IN ITEMS s4_config s5_math)
  add_executable(test_${stage_test} "${STITCH_REPO_ROOT}/tests/test_${stage_test}.cpp")
  target_include_directories(test_${stage_test} PRIVATE "${STITCH_REPO_ROOT}/tests")
  target_link_libraries(test_${stage_test} PRIVATE stitch_s4_config stitch_s5_config)
  add_test(NAME test_${stage_test} COMMAND test_${stage_test})
endforeach()

include(CheckLanguage)
check_language(CUDA)
if(CMAKE_CUDA_COMPILER)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 75)
  endif()
  enable_language(CUDA)
  foreach(stage IN ITEMS s4 s5)
    add_library(stitch_${stage} STATIC ${stage}.cu)
    target_link_libraries(stitch_${stage} PUBLIC stitch_${stage}_config)
    set_target_properties(stitch_${stage} PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
    add_executable(test_${stage}_cuda "${STITCH_REPO_ROOT}/tests/test_${stage}_cuda.cu")
    target_include_directories(test_${stage}_cuda PRIVATE "${STITCH_REPO_ROOT}/tests")
    target_link_libraries(test_${stage}_cuda PRIVATE stitch_${stage})
    set_target_properties(test_${stage}_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
    add_test(NAME test_${stage}_cuda COMMAND test_${stage}_cuda)
    set_tests_properties(test_${stage}_cuda PROPERTIES SKIP_RETURN_CODE 77)
  endforeach()
  add_executable(s4_s5_mock_driver "${STITCH_REPO_ROOT}/tests/s4_s5_mock_driver.cu")
  target_include_directories(s4_s5_mock_driver PRIVATE "${STITCH_REPO_ROOT}/tests")
  target_link_libraries(s4_s5_mock_driver PRIVATE stitch_s4 stitch_s5)
  set_target_properties(s4_s5_mock_driver PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
else()
  message(STATUS "S4/S5: no CUDA compiler; only host numerical/configuration tests are built")
endif()
