cmake_minimum_required(VERSION 3.18)

# Standalone stage build; shared root build files remain owned by Tuan.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(crossstage_s5 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  enable_testing()
endif()

get_filename_component(STITCH_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT TARGET stitch_common)
  add_library(stitch_common INTERFACE)
  target_include_directories(stitch_common INTERFACE "${STITCH_REPO_ROOT}/src")
endif()

add_library(stitch_s5_config STATIC s5_config.cpp)
target_link_libraries(stitch_s5_config PUBLIC stitch_common)
target_compile_features(stitch_s5_config PUBLIC cxx_std_17)
add_executable(test_s5_math "${STITCH_REPO_ROOT}/tests/test_s5_math.cpp")
target_include_directories(test_s5_math PRIVATE "${STITCH_REPO_ROOT}/tests")
target_link_libraries(test_s5_math PRIVATE stitch_s5_config)
add_test(NAME test_s5_math COMMAND test_s5_math)

include(CheckLanguage)
check_language(CUDA)
if(CMAKE_CUDA_COMPILER)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 75)
  endif()
  enable_language(CUDA)
  add_library(stitch_s5 STATIC s5.cu)
  target_link_libraries(stitch_s5 PUBLIC stitch_s5_config)
  set_target_properties(stitch_s5 PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_executable(test_s5_cuda "${STITCH_REPO_ROOT}/tests/test_s5_cuda.cu")
  target_include_directories(test_s5_cuda PRIVATE "${STITCH_REPO_ROOT}/tests")
  target_link_libraries(test_s5_cuda PRIVATE stitch_s5)
  set_target_properties(test_s5_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_test(NAME test_s5_cuda COMMAND test_s5_cuda)
  set_tests_properties(test_s5_cuda PROPERTIES SKIP_RETURN_CODE 77)
else()
  message(STATUS "S5: no CUDA compiler; only host numerical/configuration tests are built")
endif()
