cmake_minimum_required(VERSION 3.18)

# May be configured directly without changing the shared team's root CMakeLists.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(crossstage_s6 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  enable_testing()
endif()

get_filename_component(STITCH_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT TARGET stitch_common)
  add_library(stitch_common INTERFACE)
  target_include_directories(stitch_common INTERFACE "${STITCH_REPO_ROOT}/src")
endif()

add_library(stitch_s6_geometry STATIC geometry.cpp)
target_link_libraries(stitch_s6_geometry PUBLIC stitch_common)
target_compile_features(stitch_s6_geometry PUBLIC cxx_std_17)
add_executable(test_s6_geometry "${STITCH_REPO_ROOT}/tests/test_s6_geometry.cpp")
target_link_libraries(test_s6_geometry PRIVATE stitch_s6_geometry)
target_include_directories(test_s6_geometry PRIVATE "${STITCH_REPO_ROOT}/tests")
add_test(NAME test_s6_geometry COMMAND test_s6_geometry)

include(CheckLanguage)
check_language(CUDA)
if(CMAKE_CUDA_COMPILER)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 75)
  endif()
  enable_language(CUDA)
  add_library(stitch_s6 STATIC s6.cu)
  target_link_libraries(stitch_s6 PUBLIC stitch_s6_geometry)
  set_target_properties(stitch_s6 PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_executable(test_s6_cuda "${STITCH_REPO_ROOT}/tests/test_s6_cuda.cu")
  target_link_libraries(test_s6_cuda PRIVATE stitch_s6)
  target_include_directories(test_s6_cuda PRIVATE "${STITCH_REPO_ROOT}/tests")
  set_target_properties(test_s6_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_test(NAME test_s6_cuda COMMAND test_s6_cuda)
  set_tests_properties(test_s6_cuda PROPERTIES SKIP_RETURN_CODE 77)
else()
  message(STATUS "S6: no CUDA compiler; only geometry host tests are built")
endif()
